5. Preprocessing Pipeline¶

Numeric → median impute + scale. Categorical → mode impute + one-hot. Binary → pass-through.

In [7]:
y = df['stroke']
X = df.drop(columns=['stroke'])

num_cols = ['age','avg_glucose_level','bmi']
cat_cols = ['gender','ever_married','work_type','Residence_type','smoking_status']
bin_cols = ['hypertension','heart_disease']

numeric_pipe = Pipeline([
    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())
])
categorical_pipe = Pipeline([
    ('imputer', SimpleImputer(strategy='most_frequent')),
    ('onehot', OneHotEncoder(handle_unknown='ignore', sparse_output=False))
])
preprocessor = ColumnTransformer([
    ('num', numeric_pipe, num_cols),
    ('cat', categorical_pipe, cat_cols),
    ('bin', 'passthrough', bin_cols)
])

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.2, random_state=RNG, stratify=y
)
print(f'Train: {X_train.shape}  Test: {X_test.shape}')

Train: (4087, 10)  Test: (1022, 10)

6. Train 9 Models¶

Each model is wrapped in a pipeline with SMOTE applied only to the training fold.

In [8]:
voting_estimators = [
    ('lr', LogisticRegression(max_iter=1000, random_state=RNG)),
    ('rf', RandomForestClassifier(n_estimators=300, random_state=RNG, n_jobs=-1)),
    ('gb', GradientBoostingClassifier(random_state=RNG)),
]

models = {
    'Logistic Regression':  LogisticRegression(max_iter=1000, random_state=RNG),
    'Decision Tree':        DecisionTreeClassifier(max_depth=6, random_state=RNG),
    'Random Forest':        RandomForestClassifier(n_estimators=300, random_state=RNG, n_jobs=-1),
    'AdaBoost':             AdaBoostClassifier(n_estimators=300, random_state=RNG),
    'Gradient Boosting':    GradientBoostingClassifier(random_state=RNG),
    'XGBoost':              XGBClassifier(n_estimators=300, learning_rate=0.1, max_depth=4,
                                          random_state=RNG, eval_metric='logloss', n_jobs=-1),
    'Voting Classifier':    VotingClassifier(estimators=voting_estimators, voting='soft', n_jobs=-1),
    'SVM (RBF kernel)':     SVC(kernel='rbf', probability=True, random_state=RNG),
    'Neural Network (MLP)': MLPClassifier(hidden_layer_sizes=(64,32), activation='relu',
                                          max_iter=300, random_state=RNG, early_stopping=True),
}

results, trained, roc_data = [], {}, {}
for name, clf in models.items():
    pipe = ImbPipeline([
        ('prep', preprocessor),
        ('smote', SMOTE(random_state=RNG)),
        ('clf', clf)
    ])
    pipe.fit(X_train, y_train)
    y_pred = pipe.predict(X_test)
    y_proba = pipe.predict_proba(X_test)[:,1]
    results.append({
        'Model':     name,
        'Accuracy':  accuracy_score(y_test, y_pred),
        'Precision': precision_score(y_test, y_pred, zero_division=0),
        'Recall':    recall_score(y_test, y_pred),
        'F1-Score':  f1_score(y_test, y_pred),
        'ROC-AUC':   roc_auc_score(y_test, y_proba),
    })
    trained[name] = pipe
    roc_data[name] = roc_curve(y_test, y_proba)
    print(f'  {name:24s}  AUC={results[-1]["ROC-AUC"]:.3f}  Rec={results[-1]["Recall"]:.3f}')

12. Threshold Tuning¶

The default 0.5 threshold is rarely optimal for imbalanced healthcare data. Sweep the threshold to see how precision, recall and F1 trade off.

In [14]:
y_proba_best = best_pipe.predict_proba(X_test)[:, 1]

thresholds = np.linspace(0.05, 0.95, 91)
prec_list, rec_list, f1_list = [], [], []
for t in thresholds:
    pred = (y_proba_best >= t).astype(int)
    prec_list.append(precision_score(y_test, pred, zero_division=0))
    rec_list.append(recall_score(y_test, pred))
    f1_list.append(f1_score(y_test, pred))

best_f1_idx = int(np.argmax(f1_list))
best_f1_th  = thresholds[best_f1_idx]
hi_recall_idx = max(next(i for i, r in enumerate(rec_list) if r < 0.85) - 1, 0)
print(f'Best-F1 threshold      : {best_f1_th:.2f}  (P={prec_list[best_f1_idx]:.3f}, R={rec_list[best_f1_idx]:.3f}, F1={f1_list[best_f1_idx]:.3f})')
print(f'Recall>=0.85 threshold : {thresholds[hi_recall_idx]:.2f}  (P={prec_list[hi_recall_idx]:.3f}, R={rec_list[hi_recall_idx]:.3f})')

14. Per-Subgroup Fairness Audit¶

Does the model perform equally well across age bands, genders, and work types? Flags any subgroup where performance drops below the overall AUC.

In [18]:
test = X_test.copy()
test['y_true']  = y_test.values
test['y_proba'] = y_proba_best
test['y_pred']  = (y_proba_best >= 0.5).astype(int)
test['age_band'] = pd.cut(test['age'], bins=[0,30,50,70,120], labels=['<30','30-49','50-69','70+'])

rows = []
for col, label in [('gender','Gender'), ('age_band','Age band'), ('work_type','Work type')]:
    for group, g in test.groupby(col, observed=True):
        if g['y_true'].nunique() < 2:
            auc, rec = np.nan, np.nan
        else:
            auc = roc_auc_score(g['y_true'], g['y_proba'])
            rec = recall_score(g['y_true'], g['y_pred'])
        rows.append({'Attribute': label, 'Group': str(group),
                     'N': len(g),
                     'Stroke rate %': round(g['y_true'].mean()*100, 2),
                     'ROC-AUC': round(auc,3) if not np.isnan(auc) else None,
                     'Recall @0.5': round(rec,3) if not np.isnan(rec) else None})
fairness_df = pd.DataFrame(rows)
